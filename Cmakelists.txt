cmake_minimum_required(VERSION 3.20)

project(MyTorch)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


# =========================
# LibTorch
# =========================

set(CMAKE_PREFIX_PATH "C:/torch/libtorch")

find_package(Torch REQUIRED)


# =========================
# OpenCV
# =========================

set(OpenCV_DIR "C:/CVV/opencv/build/x64/vc16/lib")

find_package(OpenCV REQUIRED)


# =========================
# Program
# =========================

add_executable(MyTorch
    main.cpp
    src/camera.cpp
    src/image_processor.cpp
)


# =========================
# Include directories
# =========================

target_include_directories(MyTorch
    PRIVATE
    ${OpenCV_INCLUDE_DIRS}
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)


# =========================
# Libraries
# =========================

target_link_libraries(MyTorch
    PRIVATE
    ${TORCH_LIBRARIES}
    ${OpenCV_LIBS}
)


# =========================
# C++ standard
# =========================

set_property(TARGET MyTorch PROPERTY CXX_STANDARD 20)


# =========================
# Copy DLLs next to EXE
# =========================

if(WIN32)

    # LibTorch DLLs
    add_custom_command(TARGET MyTorch POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "C:/torch/libtorch/lib"
        "$<TARGET_FILE_DIR:MyTorch>"
    )

    # OpenCV DLLs
    add_custom_command(TARGET MyTorch POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "C:/CVV/opencv/build/x64/vc16/bin"
        "$<TARGET_FILE_DIR:MyTorch>"
    )

endif()
